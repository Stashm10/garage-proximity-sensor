// PIN TEST - is a GPIO damaged?
//
// BEFORE RUNNING: unplug every wire from GPIO 18, 19, 16 and 4 on the ESP32.
// Those four pins must have NOTHING connected to them.
//
// Each pin gets an internal pulldown resistor switched on, which ties it to
// ground from inside the chip. A healthy unconnected pin MUST read 0.
// Any pin reading 1 is damaged.
//
// Serial Monitor at 115200.

int pins[] = {18, 19, 16, 4};
const int n = 4;

void setup() {
  Serial.begin(115200);
  delay(400);
  Serial.println();
  Serial.println("=== PIN TEST ===");
  Serial.println("Unplug all wires from GPIO 18, 19, 16, 4 first!");
  Serial.println();

  for (int i = 0; i < n; i++) {
    pinMode(pins[i], INPUT_PULLDOWN);
  }
}

void loop() {
  Serial.print("reads:  ");
  for (int i = 0; i < n; i++) {
    Serial.print("GPIO");
    Serial.print(pins[i]);
    Serial.print("=");
    Serial.print(digitalRead(pins[i]));
    Serial.print("   ");
  }
  Serial.println("  <- every one should be 0");
  delay(500);
}
