// Distance in inches on the OLED, with a too-close warning.
//
// WIRING
//   HC-SR04:  VCC -> VIN     Gnd -> GND     Trig -> D12    Echo -> D13
//   OLED:     VCC -> 3V3     GND -> GND     SDA  -> D21    SCL  -> D22
//
// Serial Monitor at 115200.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define TRIG_PIN 12
#define ECHO_PIN 13

#define SCREEN_W   128
#define SCREEN_H   64
#define OLED_ADDR  0x3C     // some modules are 0x3D - try that if blank

#define TOO_CLOSE_INCHES 8.0f    // change this to retune the warning

Adafruit_SSD1306 display(SCREEN_W, SCREEN_H, &Wire, -1);
bool haveDisplay = false;

void setup() {
  Serial.begin(115200);
  delay(200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  digitalWrite(TRIG_PIN, LOW);

  haveDisplay = display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  if (!haveDisplay) {
    Serial.println("OLED not found at 0x3C - check wiring, or try 0x3D");
  } else {
    display.clearDisplay();
    display.setTextColor(SSD1306_WHITE);
    display.display();
  }
}

// One ping. Returns inches, or -1 if nothing came back.
float pingInches() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(3);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return -1.0f;

  // Round trip, so halve it. 0.0135205 in/us = speed of sound at 20C.
  float inches = duration * 0.0135205f / 2.0f;

  // Anything outside the sensor's real range is noise, not a reading.
  if (inches < 1.0f || inches > 120.0f) return -1.0f;
  return inches;
}

// Median of three pings. This is what kills the occasional wild value -
// one bad reading can't win a vote of three.
float readInches() {
  float a = pingInches(); delay(20);
  float b = pingInches(); delay(20);
  float c = pingInches();

  if (a < 0 || b < 0 || c < 0) return -1.0f;

  float hi = max(a, max(b, c));
  float lo = min(a, min(b, c));
  return a + b + c - hi - lo;      // whatever is left is the middle one
}

void loop() {
  float inches = readInches();

  if (inches < 0) {
    Serial.println("no target");
  } else {
    Serial.print(inches, 1);
    Serial.println(" in");
  }

  if (haveDisplay) {
    display.clearDisplay();

    if (inches < 0) {
      display.setTextSize(2);
      display.setCursor(0, 24);
      display.print("no target");

    } else if (inches < TOO_CLOSE_INCHES) {
      display.setTextSize(2);
      display.setCursor(0, 4);
      display.print("Roman!");
      display.setCursor(0, 26);
      display.print("Too Close!");
      display.setTextSize(1);
      display.setCursor(0, 52);
      display.print(inches, 1);
      display.print(" in");

    } else {
      display.setTextSize(1);
      display.setCursor(0, 0);
      display.print("DISTANCE");
      display.setTextSize(3);
      display.setCursor(0, 20);
      display.print(inches, 1);
      display.setTextSize(2);
      display.print(" in");
    }

    display.display();
  }

  delay(100);
}
