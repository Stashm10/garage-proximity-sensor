// Distance on the OLED - a single self-contained sketch.
//
// Same idea as the YouTube video, but wired for YOUR board:
//   TRIG = GPIO 17, printed TX2 on your ESP32
//   ECHO = GPIO 16, printed RX2, through the 1k/2k divider
//   OLED = GPIO 21 (SDA, printed D21) and GPIO 22 (SCL, printed D22)
//
// Needs only the libraries you already installed: Adafruit GFX and
// Adafruit SSD1306. No HCSR04 library required.
//
// Serial Monitor at 115200.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define PIN_TRIG   17     // your board prints this as TX2
#define PIN_ECHO   16     // your board prints this as RX2

#define SCREEN_W   128
#define SCREEN_H   64
#define OLED_ADDR  0x3C   // some modules are 0x3D - try that if blank

Adafruit_SSD1306 display(SCREEN_W, SCREEN_H, &Wire, -1);
bool haveDisplay = false;

void setup() {
  Serial.begin(115200);
  delay(100);

  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  digitalWrite(PIN_TRIG, LOW);

  haveDisplay = display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  if (!haveDisplay) {
    Serial.println("OLED not found at 0x3C - check wiring, or try 0x3D");
  } else {
    display.clearDisplay();
    display.setTextColor(SSD1306_WHITE);
    display.setTextSize(1);
    display.setCursor(0, 0);
    display.println("starting...");
    display.display();
  }

  Serial.println("distance + OLED test");
}

// Returns inches, or -1.0 when no echo came back.
float readInches() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(3);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);

  unsigned long dur = pulseIn(PIN_ECHO, HIGH, 25000);
  if (dur == 0) return -1.0f;

  // Round trip, so halve it. 0.0135205 in/us is the speed of sound at 20C.
  return (dur * 0.0135205f) / 2.0f;
}

void loop() {
  float inches = readInches();

  if (inches < 0) {
    Serial.println("no target");
  } else {
    Serial.print(inches, 1);
    Serial.print(" in   ");
    Serial.print(inches * 2.54f, 1);
    Serial.println(" cm");
  }

  if (haveDisplay) {
    display.clearDisplay();

    display.setTextSize(1);
    display.setCursor(0, 0);
    display.print("DISTANCE");

    display.setTextSize(3);
    display.setCursor(0, 18);
    if (inches < 0) display.print("--");
    else            display.print(inches, 1);

    display.setTextSize(1);
    display.setCursor(0, 52);
    if (inches < 0) {
      display.print("no target");
    } else {
      display.print(inches * 2.54f, 1);
      display.print(" cm");
    }

    display.display();
  }

  delay(150);
}
